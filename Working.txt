import json
import os
from jinja2 import Environment, FileSystemLoader

def build_static_html():
    # 1. Initialize environment setup
    env = Environment(loader=FileSystemLoader('templates'))
    # 2. Register Global JSON keys directly onto Jinja Environment
    with open('global.json', 'r') as f:
        global_vars = json.load(f)
    env.globals.update(global_vars) 
    
    # 3. Load local Page Specific JSON data 
    with open('page_data.json', 'r') as f:
        page_vars = json.load(f)
        
    # 4. Extract and compile File 1 (main.html)
    template = env.get_template('main.html')
    output_html = template.render(page_vars)
    
    # 5. Output compilation delivery
    os.makedirs('dist', exist_ok=True)
    with open('dist/index.html', 'w') as f:
        f.write(output_html)
        
    print("Static HTML completely generated under dist/index.html")

if __name__ == '__main__':
    build_static_html()
